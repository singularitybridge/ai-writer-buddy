TypeScript Interfaces - משימות ונקודות

export interface Challenge {
  // מזהה ייחודי
  id: string;

  // כותרת בעברית
  title: string;

  // תיאור המשימה
  description: string;

  // נקודות עבור השלמה
  points: number;

  // רמת קושי: beginner | intermediate | advanced
  difficulty: 'beginner' | 'intermediate' | 'advanced';

  // קטגוריה: crud | ai | voice | review | i18n
  category: string;

  // תגיות טכנולוגיות
  tags: string[];

  // הוראות יישום מפורטות
  instructions: string;

  // קריטריוני הצלחה
  successCriteria: string[];

  // משימות תלויות (prerequisites)
  dependencies: string[];

  // משאבים רלוונטיים
  resources: {
    title: string;
    url: string;
    type: 'docs' | 'video' | 'article';
  }[];

  // קוד לדוגמה (optional)
  codeExample?: string;

  // טיפים (hints) מדורגים
  hints: {
    level: number;
    text: string;
    pointsPenalty: number;
  }[];
}

export interface UserProgress {
  // מזהה משתמש (יכול להיות local storage בינתיים)
  userId: string;

  // סך כל הנקודות שנצברו
  totalPoints: number;

  // משימות שהושלמו
  completedChallenges: {
    challengeId: string;
    completedAt: string;
    pointsEarned: number;
    hintsUsed: number;
  }[];

  // משימות בתהליך
  inProgressChallenges: {
    challengeId: string;
    startedAt: string;
    hintsUsed: number;
  }[];

  // רמה נוכחית (calculated)
  level: number;

  // תגי הישגים
  badges: {
    id: string;
    name: string;
    earnedAt: string;
  }[];
}

export interface TechGuide {
  // מזהה ייחודי
  id: string;

  // כותרת בעברית
  title: string;

  // תיאור קצר
  description: string;

  // קטגוריה: architecture | tech-stack | patterns | tools
  category: string;

  // תוכן (Markdown format)
  content: string;

  // סדר תצוגה
  order: number;

  // תאריך עדכון אחרון
  updatedAt: string;

  // קישורים קשורים
  relatedGuides: string[];
}

export interface AIToolSetup {
  // שם הכלי: Claude Code | OpenAI Codex | Gemini CLI
  toolName: string;

  // שלבי התקנה
  setupSteps: {
    order: number;
    title: string;
    instructions: string;
    codeSnippet?: string;
  }[];

  // תצורה מומלצת
  recommendedConfig: string;

  // טיפים לשימוש
  usageTips: string[];
}

API Endpoints חדשים

# GET /api/challenges - קבלת כל המשימות
Query Parameters:
  difficulty?: 'beginner' | 'intermediate' | 'advanced'
  category?: string
  completed?: boolean

Response:
  {
    challenges: Challenge[]
  }
# GET /api/challenges/[id] - קבלת משימה ספציפית
Response:
  {
    challenge: Challenge,
    userProgress?: {
      started: boolean,
      hintsUsed: number,
      completed: boolean
    }
  }
# POST /api/progress/complete - סימון משימה כהושלמה
Body:
  {
    challengeId: string,
    hintsUsed: number
  }

Response:
  {
    pointsEarned: number,
    totalPoints: number,
    newBadges?: Badge[]
  }
# GET /api/guides - קבלת כל המדריכים
Query Parameters:
  category?: string

Response:
  {
    guides: TechGuide[]
  }
# GET /api/guides/[slug] - קבלת מדריך ספציפי
Response:
  {
    guide: TechGuide,
    relatedGuides: TechGuide[]
  }
